// ============================================================ // BRN P2P — Cliente Web3 (Ethers v5) + WebSocket // ============================================================ // ⚙️ CONFIG — ajuste após o deploy do contrato const CONFIG = { CONTRACT_ADDRESS: "0x0000000000000000000000000000000000000000", // ← COLE AQUI CHAIN_ID: 137, // Polygon Mainnet USDC_ADDRESS: "0x2791Bca1f2de4661ED88A30C99A7a9449Aa84174", DEC_USDC: 6, DEC_COTACAO: 8, // cotação salva com 8 casas EXPLORER: "https://polygonscan.com", }; const ABI_CONTRACT = [ "function executarOrdem((address criador,uint256 valorUSDC,uint256 cotacao,uint256 nonce,uint256 expiracao) _ordem, bytes _assinatura, address _comprador) external", "function cancelarOrdemOffChain(uint256 _nonce) external", "function nonceUtilizado(address,uint256) view returns (bool)", "function taxaAplicacao() view returns (uint256)" ]; const ABI_ERC20 = [ "function balanceOf(address) view returns (uint256)", "function allowance(address,address) view returns (uint256)", "function approve(address,uint256) returns (bool)", "function decimals() view returns (uint8)" ]; // Estado global let provider, signer, endereco, contrato, usdcContrato; let ws, wsReconnectTimer; let ordensMural = {}; // hash -> ordem const MAX_UINT = "0x" + "f".repeat(64); // ============================================================ // UTILITÁRIOS // ============================================================ const $ = (id) => document.getElementById(id); function paraWei(str, decimals) { const s = String(str).trim().replace(",", "."); if (!/^\d+(\.\d+)?$/.test(s)) throw new Error("Valor inválido"); const [i, f = ""] = s.split("."); const fp = (f + "0".repeat(decimals)).slice(0, decimals); return ethers.BigNumber.from(i + fp); } function deWei(big, decimals, casas = 6) { const s = ethers.utils.formatUnits(big, decimals); const n = Number(s); return n.toLocaleString("pt-BR", { minimumFractionDigits: 2, maximumFractionDigits: casas }); } function msgErro(t) { $("mensagem").innerHTML = `
❌ ${t}
`; setTimeout(() => $("mensagem").innerHTML = "", 6000); } function msgOk(t) { $("mensagem").innerHTML = `
✅ ${t}
`; setTimeout(() => $("mensagem").innerHTML = "", 5000); } function msgAviso(t) { $("mensagem").innerHTML = `
⏳ ${t}
`; } function shortAddr(a) { return a.slice(0, 6) + "..." + a.slice(-4); } // ============================================================ // INICIALIZAÇÃO // ============================================================ async function init() { if (!window.ethereum) { msgErro("Instale a MetaMask: https://metamask.io"); return; } if (CONFIG.CONTRACT_ADDRESS === "0x0000000000000000000000000000000000000000") { msgAviso("⚠️ Configure CONFIG.CONTRACT_ADDRESS no app.js"); } provider = new ethers.providers.Web3Provider(window.ethereum, "any"); // Listeners de carteira window.ethereum.on("accountsChanged", (contas) => { if (contas.length === 0) { location.reload(); return; } endereco = contas[0]; $("endereco-carteira").textContent = shortAddr(endereco); recarregar(); }); window.ethereum.on("chainChanged", () => location.reload()); // Se já autorizou antes, reconecta silenciosamente const contas = await provider.listAccounts(); if (contas.length > 0) await conectar(); // Inicia WebSocket conectarWS(); // Loop de atualização do countdown setInterval(renderizarMural, 1000); } // ============================================================ // CONECTAR CARTEIRA // ============================================================ async function conectar() { try { await provider.send("eth_requestAccounts", []); const net = await provider.getNetwork(); if (Number(net.chainId) !== CONFIG.CHAIN_ID) { msgErro(`Rede errada (chainId ${net.chainId}). Use Polygon Mainnet.`); return; } signer = provider.getSigner(); endereco = await signer.getAddress(); contrato = new ethers.Contract(CONFIG.CONTRACT_ADDRESS, ABI_CONTRACT, signer); usdcContrato = new ethers.Contract(CONFIG.USDC_ADDRESS, ABI_ERC20, signer); $("btn-conectar").style.display = "none"; $("endereco-carteira").textContent = shortAddr(endereco); $("endereco-carteira").style.display = "inline-block"; await atualizarSaldo(); renderizarMural(); renderizarMinhas(); } catch (e) { if (e.code === 4001) return; msgErro("Falha ao conectar: " + (e.message || e)); } } async function atualizarSaldo() { if (!usdcContrato || !endereco) return; try { const saldo = await usdcContrato.balanceOf(endereco); $("stat-saldo-usdc").textContent = deWei(saldo, CONFIG.DEC_USDC) + " USDC"; } catch (e) { console.error(e); } } // ============================================================ // CRIAR ORDEM (assinatura EIP-712) // ============================================================ async function criarOrdem() { if (!signer) { await conectar(); return; } if (!signer) return; const usdcStr = $("venda-usdc").value.trim(); const cotStr = $("venda-cotacao").value.trim(); const expMin = Number($("venda-expiracao").value); if (!usdcStr || !cotStr) { msgErro("Preencha todos os campos"); return; } let valorUSDC, cotacao, expiracao, nonce; try { valorUSDC = paraWei(usdcStr, CONFIG.DEC_USDC); cotacao = paraWei(cotStr, CONFIG.DEC_COTACAO); } catch (e) { msgErro("Valores inválidos"); return; } if (valorUSDC.lte(0) || cotacao.lte(0)) { msgErro("Valores devem ser positivos"); return; } // Confere saldo const saldo = await usdcContrato.balanceOf(endereco); if (saldo.lt(valorUSDC)) { msgErro(`Saldo USDC insuficiente. Você tem ${deWei(saldo, CONFIG.DEC_USDC)} USDC`); return; } // Aprova contrato (max) se necessário msgAviso("Verificando allowance..."); const allowance = await usdcContrato.allowance(endereco, CONFIG.CONTRACT_ADDRESS); if (allowance.lt(valorUSDC)) { try { msgAviso("Aprovando USDC ao contrato..."); const tx = await usdcContrato.approve(CONFIG.CONTRACT_ADDRESS, MAX_UINT); await tx.wait(); msgOk("Aprovação confirmada."); } catch (e) { if (e.code === 4001) { msgErro("Aprovação cancelada"); return; } msgErro("Falha na aprovação: " + (e.message || e)); return; } } // Nonce único baseado em timestamp (evita colisão) nonce = Math.floor(Date.now() / 1000) * 1000 + Math.floor(Math.random() * 1000); expiracao = Math.floor(Date.now() / 1000) + expMin * 60; const domain = { name: "CarteiraBRN_P2P", version: "1", chainId: CONFIG.CHAIN_ID, verifyingContract: CONFIG.CONTRACT_ADDRESS, }; const types = { Ordem: [ { name: "criador", type: "address" }, { name: "valorUSDC", type: "uint256" }, { name: "cotacao", type: "uint256" }, { name: "nonce", type: "uint256" }, { name: "expiracao", type: "uint256" }, ], }; const value = { criador: endereco, valorUSDC: valorUSDC.toString(), cotacao: cotacao.toString(), nonce: nonce.toString(), expiracao: expiracao.toString(), }; try { msgAviso("Aguardando assinatura na carteira..."); const assinatura = await signer._signTypedData(domain, types, value); // O hash usado como chave no mural é o mesmo digest EIP-712 const hash = ethers.utils._TypedDataEncoder.hash(domain, types, value); const ordem = { hash, criador: endereco, contratoAddress: CONFIG.CONTRACT_ADDRESS, valorOferecido: valorUSDC.toString(), // USDC em wei (string) valorDesejado: cotacao.toString(), // cotação * 1e8 (string) valorUSDC: valorUSDC.toString(), cotacao: cotacao.toString(), nonce: nonce.toString(), expiracao, assinatura, chainId: CONFIG.CHAIN_ID, }; msgAviso("Publicando no mural..."); enviarWS({ tipo: "publicar_ordem", ordem }); $("venda-usdc").value = ""; $("venda-cotacao").value = ""; $("total-estimado").textContent = "— BRL"; } catch (e) { if (e.code === 4001) { msgErro("Assinatura cancelada"); return; } msgErro("Erro ao assinar: " + (e.message || e)); } } // ============================================================ // EXECUTAR ORDEM (comprador) // ============================================================ async function executarOrdem(hash) { const ordem = ordensMural[hash]; if (!ordem) { msgErro("Ordem não encontrada"); return; } if (!signer) { await conectar(); return; } if (!signer) return; if (ordem.criador.toLowerCase() === endereco.toLowerCase()) { msgErro("Você não pode comprar sua própria ordem"); return; } const totalBRL = (Number(ordem.valorUSDC) / 1e6) * (Number(ordem.cotacao) / 1e8); if (!confirm( `Confirmar compra?\n\n` + `USDC recebido: ${deWei(ethers.BigNumber.from(ordem.valorUSDC), 6)} USDC\n` + `Cotação: ${(Number(ordem.cotacao)/1e8).toFixed(2)} BRL/USDC\n` + `Total: ~R$ ${totalBRL.toFixed(2)}\n\n` + `⚠️ Combine o pagamento em BRL com o vendedor ANTES de confirmar.` )) return; try { msgAviso("Enviando transação para executar ordem..."); const structOrdem = { criador: ordem.criador, valorUSDC: ordem.valorUSDC, cotacao: ordem.cotacao, nonce: ordem.nonce, expiracao: ordem.expiracao, }; const tx = await contrato.executarOrdem(structOrdem, ordem.assinatura, endereco); msgAviso(`Tx enviada: ${tx.hash.slice(0, 12)}... aguardando confirmação`); await tx.wait(); msgOk("Ordem executada com sucesso! USDC recebido."); enviarWS({ tipo: "ordem_executada", hash, txHash: tx.hash }); await atualizarSaldo(); } catch (e) { if (e.code === 4001) { msgErro("Transação cancelada"); return; } let detalhe = e.message || String(e); if (e.data && e.data.message) detalhe = e.data.message; msgErro("Falha: " + detalhe); } } // ============================================================ // CANCELAR ORDEM (vendedor) // ============================================================ async function cancelarOrdem(hash) { const ordem = ordensMural[hash]; if (!ordem) { msgErro("Ordem não encontrada"); return; } if (!signer) return; if (ordem.criador.toLowerCase() !== endereco.toLowerCase()) { msgErro("Só o criador pode cancelar"); return; } if (!confirm("Cancelar esta ordem? A nonce será marcada como usada on-chain.")) return; try { msgAviso("Cancelando on-chain..."); const tx = await contrato.cancelarOrdemOffChain(ordem.nonce); await tx.wait(); msgOk("Ordem cancelada on-chain."); enviarWS({ tipo: "cancelar_ordem", hash }); } catch (e) { if (e.code === 4001) { msgErro("Transação cancelada"); return; } msgErro("Erro: " + (e.message || e)); } } // ============================================================ // WEBSOCKET // ============================================================ function enviarWS(payload) { if (!ws || ws.readyState !== WebSocket.OPEN) { msgErro("Sem conexão com o servidor. Tentando reconectar..."); conectarWS(); return; } ws.send(JSON.stringify(payload)); } function conectarWS() { if (ws && (ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING)) return; const proto = location.protocol === "https:" ? "wss:" : "ws:"; const url = `${proto}//${location.host}/ws`; try { ws = new WebSocket(url); } catch (e) { agendarReconexao(); return; } ws.onopen = () => { $("ws-dot").className = "dot dot-on"; $("ws-status").textContent = "Conectado"; $("status-info").textContent = "Mural sincronizado"; }; ws.onclose = () => { $("ws-dot").className = "dot dot-off"; $("ws-status").textContent = "Desconectado"; agendarReconexao(); }; ws.onerror = () => { try { ws.close(); } catch (_) {} }; ws.onmessage = (ev) => { let msg; try { msg = JSON.parse(ev.data); } catch { return; } switch (msg.tipo) { case "snapshot": ordensMural = {}; (msg.ordens || []).forEach(o => { ordensMural[o.hash] = o; }); renderizarMural(); renderizarMinhas(); break; case "nova_ordem": ordensMural[msg.ordem.hash] = msg.ordem; renderizarMural(); renderizarMinhas(); break; case "ordem_cancelada": case "ordem_executada": case "ordem_expirada": delete ordensMural[msg.hash]; renderizarMural(); renderizarMinhas(); break; case "erro": msgErro("Servidor: " + msg.msg); break; case "pong": break; } }; } function agendarReconexao() { if (wsReconnectTimer) clearTimeout(wsReconnectTimer); wsReconnectTimer = setTimeout(conectarWS, 3000); } // ============================================================ // RENDERIZAÇÃO // ============================================================ function renderizarMural() { const tbody = $("lista-ordens"); const agora = Math.floor(Date.now() / 1000); const lista = Object.values(ordensMural) .filter(o => Number(o.expiracao) > agora) .sort((a, b) => Number(b.cotacao) - Number(a.cotacao)); $("stat-total").textContent = lista.length; if (lista.length === 0) { tbody.innerHTML = ` Nenhuma ordem ativa. Seja o primeiro a criar! 🚀 `; return; } tbody.innerHTML = lista.map(o => { const usdcN = Number(o.valorUSDC) / 1e6; const cotN = Number(o.cotacao) / 1e8; const total = usdcN * cotN; const seg = Number(o.expiracao) - agora; const mm = String(Math.floor(seg / 60)).padStart(2, "0"); const ss = String(seg % 60).padStart(2, "0"); const souEu = endereco && o.criador.toLowerCase() === endereco.toLowerCase(); const acao = souEu ? `` : ``; return ` ${shortAddr(o.criador)} ${souEu ? '
você
' : ""} ${usdcN.toFixed(2)} ${cotN.toFixed(2)} BRL R$ ${total.toFixed(2)} ${mm}:${ss} ${acao} `; }).join(""); } function renderizarMinhas() { if (!endereco) { $("minhas-lista").innerHTML = "Conecte a carteira para ver suas ordens."; return; } const agora = Math.floor(Date.now() / 1000); const minhas = Object.values(ordensMural) .filter(o => o.criador.toLowerCase() === endereco.toLowerCase() && Number(o.expiracao) > agora); if (minhas.length === 0) { $("minhas-lista").innerHTML = "Você não tem ordens ativas."; return; } $("minhas-lista").innerHTML = minhas.map(o => { const usdcN = Number(o.valorUSDC) / 1e6; const cotN = Number(o.cotacao) / 1e8; const total = usdcN * cotN; const seg = Number(o.expiracao) - agora; const mm = String(Math.floor(seg / 60)).padStart(2, "0"); const ss = String(seg % 60).padStart(2, "0"); return `
${usdcN.toFixed(2)} USDC @ ${cotN.toFixed(2)}
R$ ${total.toFixed(2)} • expira em ${mm}:${ss}
`; }).join(""); } // ============================================================ // TOTAL ESTIMADO EM TEMPO REAL // ============================================================ function atualizarTotal() { const usdc = Number($("venda-usdc").value); const cot = Number($("venda-cotacao").value); if (usdc > 0 && cot > 0) { $("total-estimado").textContent = "R$ " + (usdc * cot).toFixed(2); } else { $("total-estimado").textContent = "— BRL"; } } // ============================================================ // EVENTOS // ============================================================ document.addEventListener("DOMContentLoaded", () => { $("btn-conectar").addEventListener("click", conectar); $("btn-criar").addEventListener("click", criarOrdem); $("venda-usdc").addEventListener("input", atualizarTotal); $("venda-cotacao").addEventListener("input", atualizarTotal); document.querySelectorAll(".tab-btn").forEach(btn => { btn.addEventListener("click", () => { document.querySelectorAll(".tab-btn").forEach(b => b.classList.remove("active")); btn.classList.add("active"); const tab = btn.dataset.tab; $("tab-criar").style.display = tab === "criar" ? "block" : "none"; $("tab-minhas").style.display = tab === "minhas" ? "block" : "none"; if (tab === "minhas") renderizarMinhas(); }); }); init(); }); // Expor funções usadas em onclick window.executarOrdem = executarOrdem; window.cancelarOrdem = cancelarOrdem;
